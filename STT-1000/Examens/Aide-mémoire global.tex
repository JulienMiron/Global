\subsection*{Théorie des probabilités}

\begin{itemize}
\item {\bf Quelques propriétés d'une probabilité:}
\begin{itemize}

\item $A$ et $B$ sont mutuellement exclusifs si et seulement si un seul des événements peut se produire à la fois. 

\item $A$ et $B$ sont indépendants si et seulement si la probabilité 
$\mathbb{P}[A \cap B] =\mathbb{P}[A] \, \mathbb{P}[B]$.



\item $\mathbb{P}[A\cup B\cup C]=\mathbb{P}[A] + \mathbb{P}[B]
+ \mathbb{P}[C] - \mathbb{P}[A \cap B] - \mathbb{P}[B \cap C]
- \mathbb{P}[A \cap C]
+\mathbb{P}[A \cap B \cap C]$.
\end{itemize}

\item {\bf Techniques de dénombrement:}
\vspace{-0.2cm}
\begin{align*}
    P_r^n = \frac{n!}{(n-r)!} & & C_r^n ={n \choose r} = \frac{n!}{r!(n-r)!}
\end{align*}


\item {\bf Probabilités conditionnelles:}

Si $B_1, B_2, ..., B_k$ forment une partition de $\mathcal{U}$,
c'est-\`a-dire si $B_1, B_2, ..., B_k$ sont mutuellement exclusifs
et tels que $\bigcup\limits_{i=1}^k B_i=\mathcal{U}$, alors
\begin{itemize}

\item[$\circ$]
$\displaystyle
\mathbb{P}[A] = \sum_{i=1}^{k}\mathbb{P}[A|B_i]\,\mathbb{P}[B_i]
$.
~~(loi des probabilités totales)

\item[$\circ$]
$\displaystyle
\mathbb{P}[B_j\mid A] =
\frac{\mathbb{P}[A\mid B_j]\,\mathbb{P}[B_j]}{\mathbb{P}[A]}
=\frac{\mathbb{P}[A\mid B_j]\,\mathbb{P}[B_j]}{\sum_{i=1}^{k}\mathbb{P}[A|B_i]\,\mathbb{P}[B_i]}
%{\displaystyle\sum_{i=1}^{k}\mathbb{P}[A\mid B_i]\,\mathbb{P}[B_i]}
$.
~~(théorème de Bayes)

\end{itemize}

\end{itemize}

\subsection*{Le théorème central limite}

\vspace{0.1in}
\begin{itemize}
\item
Si $X_1,\hdots, X_n$ sont des variables aléatoires
indépendantes et identiquement distribuées, avec moyenne $\mu$
et variance $\sigma^2$ et si $n$ est assez grand
($n\ge  30$), alors
\vspace{-0.2cm}
\begin{align*}
    \sum\limits_{j=1}^n X_j\approx \mathcal{N}(n\mu, n\sigma^2) & & \overline{X}\approx \mathcal{N}(\mu,\sigma^2/n)
\end{align*}

\item
Si les $X_j$ sont des variables aléatoires
discrètes, on utilise la correction pour la continuité:
\begin{eqnarray*}
\mathbb{P}(X \leq b) &\cong& \mathbb{P}\left(X < b+\frac12\right), \quad {\mathbb
P}(X < b) \cong \mathbb{P}\left(X < b - \frac12\right), \\ \mathbb{P}(X \geq a) &\cong& \mathbb{P}\left(X
> a-\frac12\right), \quad \mathbb{P}(X > a) \cong \mathbb{P}\left(X > a+\frac12\right).
\end{eqnarray*}
\end{itemize}


\subsection*{Lois discrètes bidimensionnelles}

\begin{itemize}
\item {\bf Lois marginales} : $ p_X(x)=  \sum_y p(x,y) \mbox{ et }
p_Y(y) = \sum_x p(x,y) $

\item {\bf Lois conditionnelles} : $\displaystyle p_{X | Y=y}(x) =  \frac{p(x,y)}{p_Y(y)} \mbox{ et }  p_{Y | X=x}(y)=  \frac{p(x,y)}{p_X(x)}$

\item {\bf Espérance conditionnelle} : $E(X|Y=y) = \sum_x x p_{X|Y=y}(x) \mbox{ et } E(Y|X=x) = \sum_y y p_{Y|X=x}(y) $

\item {\bf Indépendance}

\noindent $X$ et $Y$ sont indépendantes si et seulement si $p(x,y)
= p_X(x) p_Y(y)$ pour tout $(x,y)$.
\item {\bf Covariance et Corrélation}

$$\mbox{Cov}(X,Y) = E\{(X-E[X])(Y-E[Y])\} = E[XY] - E[X]E[Y]$$
$${\rho}_{(X,Y)} =
\frac{\mbox{Cov(X,Y)}}{\sqrt{\mbox{Var}(X)\mbox{Var}(Y)}} \quad
\quad -1 \le {\rho}_{(X,Y)} \le 1 $$

\end{itemize}

\subsection*{Régression linéaire}


\vspace{0.5cm}

On observe $(x_1,y_1), (x_2, y_2),..., (x_n,y_n)$ et on
suppose que $y_1, y_2,..., y_n$ sont les réalisations de
variables aléatoires indépendantes $Y_1, Y_2,..., Y_n$ telles
que pour chaque $i$ la loi de $Y_i$ est $\mathcal{N}(\beta_0 + \beta_1 x_i,
\sigma^2)$.

\vspace{0.2 in}

\textbf{Quantités utiles:} 

\vspace{-0.7 cm}
\begin{align*}
S_{xy}=\sum\limits_{i=1}^n (x_i-\overline
x)(y_i-\overline y) =  \sum\limits_{i=1}^n x_iy_i - n\;\overline
x\;\overline y & & &
S_{xx} =\sum\limits_{i=1}^n (x_i-\overline x)^2 =
\sum\limits_{i=1}^n x_i^2 - n\;\overline x\;^2\\
S_{yy} =\sum\limits_{i=1}^n (y_i-\overline y)^2 =
\sum\limits_{i=1}^n y_i^2 - n\;\overline y\;^2 & & &
R^2 = \hat{\beta}_1^2 \frac{S_{xx}}{S_{yy}}
\end{align*}

\textbf{Estimateurs des paramètres:}

$$ \bullet \; \; \hat{\beta}_1 =\frac{S_{xy}}{S_{xx}} \;\;\;\;\;\; \;\;\;\;\;\; \bullet \;\; \hat{\beta}_0 = \overline{y} - \hat{\beta}_1 \overline{x} $$
$$ \bullet \;\; \hat{\sigma}^2 = \frac{1}{n-2} \sum_{i=1}^n (e_i)^2 = \frac{1}{n-2} \sum_{i=1}^n (y_i - \hat{y}_i)^2  = \frac{1}{n-2} \sum_{i=1}^n (y_i - (\hat{\beta}_0 + \hat{\beta}_1 x_i))^2$$

\vspace{0.2in}
\textbf{Test d'hypothèse pour $\beta_1$:}
\vspace{0.1in}

Hypothèse nulle:  $H_0:  \beta_1= \beta_{1,0}$

Statistique du test sous $H_0$ : 
$$ T_0 = \frac{\hat{\beta}_1 - \beta_{1,0}}{\sqrt{\hat{\sigma}^2/S_{xx}}} \sim t_{n-2}$$

\vspace{0.05 in}
On rejette si $H_0$ versus $H_1 : \beta_1 > \beta_{1,0}$ si $t_0 > t_{\alpha, n-2}$.\\
