\section*{Th\'eorie des probabilit\'es}

Soit ${\cal S}$, l'espace \'echantillonnal et $A,B,C\in{\cal{S}}$,
des ensembles.
\begin{itemize}
\item {\bf Op\'erations sur les ensembles:}

\begin{tabular}{l r l}
 - le {\it compl\'ementaire} de A:& $\overline A$&$=\;\{x:\,x\in {\mathcal{S}} \;\mbox{ et } x \notin
 A\}$\\
 - l' {\it intersection} de A et B:& $A\cap B$&$=\;\{x:\,x\in A\;\mbox{ et } x\in
 B\}$\\
 - l' {\it union} de A et B:& $A\cup B$&$=\;\{x:\,x\in A\;\mbox{ ou } x\in B\;$ (ou les deux)\}
\end{tabular}

\item {\bf Propri\'et\'es des op\'erations:}

\begin{tabular}{l r l}
 - Lois de DeMorgan:  & $\overline {A \cup B}$&$=\;\overline A \cap \overline B$\\
                      & $\overline {A \cap B}$&$=\;\overline A \cup \overline B$\\
 - Lois associatives: & $A\cup (B\cup C)$&$=\;(A\cup B)\cup C$\\
                      & $A\cap (B\cap C)$&$=\;(A\cap B)\cap C$\\
 - Lois distributives:& $A\cup (B\cap C)$&$=\;(A\cup B)\cap (A\cup C)$\\
                      & $A\cap (B\cup C)$&$=\;(A\cap B)\cup (A\cap C)$\\
\end{tabular}

\item {\bf Propri\'et\'es d'une probabilit\'e:}
\begin{itemize}
\item $0 \le {\mathbb P}[A] \le 1$

\item ${\mathbb P}[\emptyset] = 0$ et ${\mathbb P}[{\cal S}] = 1$.

\item $A$ et $B$ sont mutuellement exclusifs si et seulement si $A\cap B = \emptyset$.

\item $A$ et $B$ sont ind\'ependants si et seulement si
${\mathbb P}[A \cap B] ={\mathbb P}[A] \, {\mathbb P}[B]$.

\item Si $A_1, A_2, A_3, ...$ sont mutuellement exclusifs, alors ${\mathbb
P}[\bigcup_{i} A_i] = \sum_{i}{\mathbb P}[A_i]$.

\item ${\mathbb P}[A] = 1 - {\mathbb P}[\overline{A}]$.
%~~(on note $\bar{A}$ ou $A^c$ le compl\'ementaire de $A$).

\item Si $A \subset B$, alors ${\mathbb P}[A] \le {\mathbb P}[B]$.

\item ${\mathbb P}[A \cup B] = {\mathbb P}[A] + {\mathbb P}[B]
- {\mathbb P}[A \cap B]$.

\item ${\mathbb P}[A\cup B\cup C]={\mathbb P}[A] + {\mathbb P}[B]
+ {\mathbb P}[C] - {\mathbb P}[A \cap B] - {\mathbb P}[B \cap C]
- {\mathbb P}[A \cap C]
+{\mathbb P}[A \cap B \cap C]$.

\item Dans le cas \'equiprobable (c'est-\`a-dire le cas o\`u ${\cal S}$ est
un ensemble fini et o\`u chaque \'el\'ement de ${\cal S}$ a la
m\^eme probabilit\'e de se r\'ealiser), on a ${\mathbb P}[A] =
\mbox{n($A$)}/\mbox{n(${\cal S}$)}$.
%${\mathbb P}[A] = \mbox{Cardinal($A$)}/\mbox{Cardinal(${\cal S}$)}$.

\end{itemize}

\item {\bf Techniques de d\'enombrement:}

\begin{tabular}{l l}
- {\it Permutations} :&$\circ$ il existe $n!$ arrangements ordonn\'es de $n$ \'el\'ements distincts;\\
                    & $\circ$ il existe $\displaystyle P_r^n = \frac{n!}{(n-r)!}$ fa\c cons de choisir
                    $r$ objets parmi $n$ objets \\
                    & en tenant compte de l'ordre. \\
-  {\it Combinaisons} :&$\circ$ il existe $\displaystyle C_r^n ={n \choose r} = \frac{n!}{r!(n-r)!}$ fa\c cons de choisir
                    $r$ objets parmi \\
                    &  $n$ objets sans tenir compte de l'ordre.
\end{tabular}

\newpage
\item {\bf Probabilit\'es conditionnelles:}

\begin{itemize}
\item
$
\begin{displaystyle}
{\mathbb P}[A\mid B] = \frac{{\mathbb P}[A \cap B]}{{\mathbb P}[B]}
\end{displaystyle}
$.


\item ${\mathbb P}[A \cap B] \;=\; {\mathbb P}[A]\,{\mathbb P}[B\mid A]
\;=\;{\mathbb P}[B]\,{\mathbb P}[A\mid B]$ (r\`egle de multiplication)

%${\mathbb P}[A \cap B \cap C] = {\mathbb P}[A]\,{\mathbb P}[B\mid A]\,{\mathbb P}[C\mid A \cap B]$.


\item Si $B_1, B_2, ..., B_k$ forment une partition de ${\cal S}$,
c'est-\`a-dire si $B_1, B_2, ..., B_k$ sont mutuellement exclusifs
et tels que $\bigcup\limits_{i=1}^k B_i={\cal S}$, alors
\begin{itemize}

\item[$\circ$]
$\displaystyle
{\mathbb P}[A] = \sum_{i=1}^{k}{\mathbb P}[A|B_i]\,{\mathbb P}[B_i]
$.
~~(loi des probabilit\'es totales)

\item[$\circ$]
$\displaystyle
{\mathbb P}[B_j\mid A] =
\frac{{\mathbb P}[A\mid B_j]\,{\mathbb P}[B_j]}{{\mathbb P}[A]}
=\frac{{\mathbb P}[A\mid B_j]\,{\mathbb P}[B_j]}{\sum_{i=1}^{k}{\mathbb P}[A|B_i]\,{\mathbb P}[B_i]}
%{\displaystyle\sum_{i=1}^{k}{\mathbb P}[A\mid B_i]\,{\mathbb P}[B_i]}
$.
~~(th\'eor\`eme de Bayes)

\end{itemize}
\end{itemize}
\end{itemize}

{\small
\begin{align*}
{\mathbb E}[aX+b]&=a{\mathbb E}[X]+b \\
{\mathbb Var}[aX+b]&=a^{2}{\mathbb Var}[X]  \\
V(aX+bY) &= a^{2}V(X) + b^{2}V(Y) +2ab\mbox{Cov}(X, Y)
\end{align*}
}

 {\small
\begin{align*}
E[Y]&=a_0+\sum_{i=1}^na_iE[X_i]\\

\end{align*}

\section*{Lois discr\`etes bidimensionnelles}

$$p(x,y) = P(\{X=x\} \cap \{Y=y\}) = P(X=x, Y=y)$$
\begin{itemize}
\item {\bf lois marginales}
\begin{eqnarray*}
p_X(x) &=& P(X=x) = \sum_y p(x,y) \\
p_Y(y) &=& P(Y=y) = \sum_x p(x,y)
\end{eqnarray*}
\item {\bf lois conditionnelles}
\begin{eqnarray*}
p_{X | Y=y}(x) &=& P(X=x | Y=y) = \frac{p(x,y)}{p_Y(y)} \\
p_{Y | X=x}(y)&=& P(Y=y | X=x) = \frac{p(x,y)}{p_X(x)}
\end{eqnarray*}
\item {\bf Esp\'erance conditionnelle}
\begin{eqnarray*}
E(X|Y=y) &=& \sum_x x p_{X|Y=y}(x) = \frac{\sum_x x p(x,y)}{p_Y(y)} \\
E(Y|X=x) &=& \sum_y y p_{Y|X=x}(y) = \frac{\sum_y y p(x,y)}{p_X(x)}
\end{eqnarray*}
\item {\bf Esp\'erance de $g(X,Y)$}
$$E[g(X,Y)] = \sum_{x,y} g(x,y) p(x,y)$$

\item {\bf Ind\'ependance}

\noindent $X$ et $Y$ sont ind\'ependantes si et seulement si $p(x,y)
= p_X(x) p_Y(y)$ pour tout $(x,y)$.
\item {\bf Covariance et Corr\'elation}

$$\mbox{Cov}(X,Y) = E\{(X-E[X])(Y-E[Y])\} = E[XY] - E[X]E[Y]$$
$${\rho}_{(X,Y)} =
\frac{\mbox{Cov(X,Y)}}{\sqrt{\mbox{Var}(X)\mbox{Var}(Y)}} \quad
\quad -1 \le {\rho}_{(X,Y)} \le 1 $$

\end{itemize}

\section*{Collecte de données}


\textbf{Échantillonnage probabiliste:} Tous les individus de la population ont une probabilité non-nulle d'être sélectionnés pour le sondage, et celle-ci est connue. \\

\textbf{Expérience statistique:} Lorsqu'on attribue au hasard un traitement aux individus pour étudier l'effet de celui-ci. \\

\textbf{Sources d'erreur dans une collecte de données :} erreur d'échantillonnage, erreur de couverture, erreur de mesure, erreur de non-réponse.

\textbf{Notes:}
\begin{itemize}
\item Pour les tableaux de fr\'equences et les donn\'ees group\'ees, $n_j$ d\'enote
le nombre d'observations pour la $j^e$ valeur de la variable
ou le $j^e$ intervalle et $y_j$ d\'enote le point milieu du $j^e$ intervalle, $j = 1, \cdots, k$.

\item
On rappelle que
$$
\sum\limits_{i=1}^n (x_i-\bar{x})^2=\sum\limits_{i=1}^n
x_i^2-n\bar{x}^2 \; \; \; \; \; \;  \mbox{ et } \; \; \; \; \; \;  \sum\limits_{j=1}^k n_j
(x_j-\bar{x})^2= \sum\limits_{j=1}^n n_j\, x_j^2-n\bar{x}^2.
$$
\end{itemize}
