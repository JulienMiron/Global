\question\label{1FQCM}

Une entreprise fabrique des palettes de chocolat et on s'intéresse à la quantité moyenne de sucre dans une palette.  On choisit au hasard 16 palettes. Dans l'échantillon, la quantité moyenne de sucre par palette est de $5,4 \,g$ et la variance échantillonnale est de $4 \, g^2$. On suppose que $X$, la quantité de sucre dans une palette, est une variable aléatoire qui suit une loi normale de moyenne $\mu$. On fait le test suivant, au seuil $\alpha=10\%$,
\begin{align*}
	H_0 &: \mu = 5 \cr
	H_1 &: \mu > 5.
\end{align*}

\begin{parts}

\part[3] Quelle est la valeur de la statistique du test $t_0$?

\medskip
\begin{oneparcheckboxes}
    \choice 1,645
    \choice 1,600
    \CorrectChoice 0,800
    \choice 1,341
    \choice 1,753 \medskip
\end{oneparcheckboxes} 

\part[2] On nous indique que le seuil observé du test est 0,218. Choisissez l'énoncé qui est vrai.

\medskip
	\begin{checkboxes}
	    \choice Au seuil $\alpha=10\%$, la quantité moyenne de sucre dans une palette est significativement supérieure à $5 \, g$.
		\CorrectChoice Au seuil $\alpha=10\%$, la quantité moyenne de sucre dans une palette n'est pas significativement supérieure à $5 \, g$.
		\choice Au seuil $\alpha=10\%$, la variance de la quantité de sucre dans une palette est significativement supérieure à $5 \, g$.
		\choice Au seuil $\alpha=10\%$, on ne peut pas dire si la quantité moyenne de sucre dans une palette est significativement différente de $5 \, g$.\medskip
	\end{checkboxes}

\end{parts}

	\begin{solution}
  (a) $$ t_0 = \frac{\overline{x}-\mu_0}{\sqrt{S^2/n}} = \frac{ 5,4- 5}{\sqrt{4/16}} = \frac{0,4}{0,5} = 0,800 $$ \\
  Erreurs possibles: 
  $ t_0 = \frac{ 5,4- 5}{{4/16}} = 1,6 $
  $ t_{n-1, \alpha} = t_{15; 0,10} = 1,341 $
  $ t_{n-1, \alpha/et 2} = t_{15; 0,05} = 1,753 $
  $ z_{\alpha/2 = z_{0,05} = 1,645$
  \\
 (b) $$ \alpha^* =  P( t_{n-1} > t_0 ) = P( t_{15} > 0,8 ) =  0.218099  > \alpha = 0,10$$
		(En R: pt(0.8, df=15, lower.tail=FALSE) )
		On ne rejette pas $H_0$. Réponse: 	B.
	\end{solution}

\question[3]\label{probabilite} 

On considère une expérience aléatoire dont deux événements  $A$ et $B$ sont tels que:\\ $P(A) = 0.3 \text{ et } P(A \cup B) = 0.7$.\\ 

Calculer $P(\overline{B}|\overline{A})$. Justifiez votre réponse.\\

\emptybox{15cm}

\begin{solution}
\begin{eqnarray*}
   P(\overline{B}|\overline{A})      &=&  \dfrac{P(\overline{B}\cap \overline{A})}{P(\overline{A})}\\
                                    &=&  \dfrac{P(\overline{B\cup A})}{1-P(A)}\\
                                    &=&  \dfrac{1-P(B\cup A)}{1-P(A)}\\
                                    &=&  \dfrac{1-0.7}{1-0.3}\\
                                    &=&  \dfrac{3}{7} .
\end{eqnarray*}
\end{solution}
