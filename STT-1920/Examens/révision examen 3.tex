\part[2] En test à 1 échantillon, si la valeur-$p$ vaut $0{,}12$ et $\alpha=0{,}05$, on...
\begin{oneparcheckboxes}
\correctchoice ne rejette pas $H_0$.
\choice rejette $H_0$.
\choice ne peut rien conclure.
\end{oneparcheckboxes}
\begin{solution}
Comme $0{,}12>0{,}05$, on ne rejette pas $H_0$.
\end{solution}

\part[2] En ANOVA à 1 facteur, l'hypothèse nulle est...
\begin{oneparcheckboxes}
\correctchoice $\mu_1=\mu_2=\cdots=\mu_a$.
\choice $\sigma_1^2=\sigma_2^2=\cdots=\sigma_a^2$.
\choice $p_1=p_2=\cdots=p_a$.
\end{oneparcheckboxes}
\begin{solution}
Le test $F$ global d'ANOVA compare l'égalité des moyennes.
\end{solution}

\part[2] En régression linéaire simple, si $\beta_1<0$, alors...
\begin{oneparcheckboxes}
\correctchoice la relation linéaire moyenne est décroissante.
\choice les variables sont indépendantes.
\choice la variance de $Y$ est négative.
\end{oneparcheckboxes}
\begin{solution}
$\beta_1$ est la pente de la droite moyenne.
\end{solution}

\part[2] Pour un khi-carré d'indépendance avec un tableau $3\times 2$, le nombre de ddl vaut...
\begin{oneparcheckboxes}
\choice 1
\correctchoice 2
\choice 3
\end{oneparcheckboxes}
\begin{solution}
$\nu=(r-1)(c-1)=(3-1)(2-1)=2$.
\end{solution}
\end{parts}

\question[8] Test à 1 échantillon.
On observe un échantillon normal de taille $n=16$ avec $\bar{x}=52$ et $s=8$.
On veut tester $H_0:\mu=50$ contre $H_1:\mu>50$ au seuil $5\%$.
\begin{parts}
\part[3] Calculez la statistique de test.
\begin{solution}
\[
T_0=\frac{\bar{x}-\mu_0}{s/\sqrt{n}}=\frac{52-50}{8/4}=\frac{2}{2}=1.
\]
\end{solution}

\part[3] Donnez la région critique.
\begin{solution}
On rejette $H_0$ si $T_0>t_{0{,}95;15}\approx 1{,}753$.
\end{solution}

\part[2] Conclusion.
\begin{solution}
Comme $1<1{,}753$, on ne rejette pas $H_0$ au seuil de $5\%$.
\end{solution}
\end{parts}

\question[8] ANOVA à 1 facteur.
Trois traitements sont comparés avec un total de 18 observations. La table partielle est:
\begin{verbatim}
Source        SS      ddl      MS      F0
Traitement   24.0      2      ???     ???
Erreur       30.0     15      ???
\end{verbatim}
\begin{parts}
\part[3] Complétez $MS_{\text{trt}}$ et $MS_E$.
\begin{solution}
$MS_{\text{trt}}=24/2=12$ et $MS_E=30/15=2$.
\end{solution}

\part[3] Calculez $F_0$.
\begin{solution}
$F_0=MS_{\text{trt}}/MS_E=12/2=6$.
\end{solution}

\part[2] Au seuil de 5\%, si $F_{0{,}05;2,15}=3{,}68$, concluez.
\begin{solution}
Comme $6>3{,}68$, on rejette $H_0$: au moins une moyenne diffère.
\end{solution}
\end{parts}

\question[8] Régression linéaire simple.
On donne les résultats suivants: $n=10$, $\bar{x}=4$, $\bar{y}=13$, $S_{XX}=20$, $S_{XY}=30$, et $R^2=0{,}81$.
\begin{parts}
\part[3] Calculez $\hat{\beta}_1$ et $\hat{\beta}_0$.
\begin{solution}
\[
\hat{\beta}_1=\frac{S_{XY}}{S_{XX}}=\frac{30}{20}=1{,}5,
\quad
\hat{\beta}_0=\bar{y}-\hat{\beta}_1\bar{x}=13-1{,}5(4)=7.
\]
Donc $\hat{y}=7+1{,}5x$.
\end{solution}

\part[2] Donnez la prédiction ponctuelle pour $x=6$.
\begin{solution}
$\hat{y}(6)=7+1{,}5(6)=16$.
\end{solution}

\part[3] Interprétez $R^2=0{,}81$.
\begin{solution}
$81\%$ de la variabilité de $Y$ est expliquée par la relation linéaire avec $X$.
\end{solution}
\end{parts}

\question[8] Khi-carré d'ajustement.
Une machine produit quatre types de capsules. 

Le fabricant annonce les proportions
$0{,}25, 0{,}25, 0{,}25, 0{,}25$. Sur $n=80$ capsules observées, on obtient
$O=(16,24,20,20)$.
\begin{parts}
\part[2] Donnez les fréquences attendues sous $H_0$.
\begin{solution}
$E_i=80(0{,}25)=20$ pour chaque catégorie.
\end{solution}

\part[4] Calculez $D_0$.
\begin{solution}
\[
D_0=\sum\frac{(O_i-E_i)^2}{E_i}
=\frac{(16-20)^2}{20}+\frac{(24-20)^2}{20}+\frac{(20-20)^2}{20}+\frac{(20-20)^2}{20}
=0{,}8+0{,}8=1{,}6.
\]
\end{solution}

\part[2] Combien de ddl?
\begin{solution}
$\nu=k-1=4-1=3$.
\end{solution}
\end{parts}

\question[8] Khi-carré d'indépendance.
On observe le tableau suivant:
\begin{center}
\begin{tabular}{c|cc|c}
 & Oui & Non & Total\\
\hline
Groupe A & 30 & 20 & 50\\
Groupe B & 10 & 40 & 50\\
\hline
Total & 40 & 60 & 100
\end{tabular}
\end{center}
\begin{parts}
\part[2] Formulez $H_0$ et $H_1$.
\begin{solution}
$H_0$: les deux variables sont indépendantes.
$H_1$: elles ne sont pas indépendantes.
\end{solution}

\part[3] Calculez les fréquences attendues.
\begin{solution}
Pour A-Oui: $E=50\times40/100=20$.
Pour A-Non: $E=50\times60/100=30$.
Pour B-Oui: $20$.
Pour B-Non: $30$.
\end{solution}

\part[3] Calculez $D_0$.
\begin{solution}
\[
D_0=\frac{(30-20)^2}{20}+\frac{(20-30)^2}{30}+\frac{(10-20)^2}{20}+\frac{(40-30)^2}{30}
=5+3{,}33+5+3{,}33=16{,}66.
\]
\end{solution}
\end{parts}

\question[8] Test à 1 échantillon - objectif: prendre une décision formelle.
Un laboratoire affirme que la concentration moyenne d'un biomarqueur est de $100$ unités.
Sur un échantillon aléatoire de $n=25$ patients, on observe $\bar{x}=104$ et $s=10$.
On veut tester au seuil $5\%$:
\[
H_0:\mu=100\qquad\text{contre}\qquad H_1:\mu>100.
\]
\begin{parts}
\part[2] Calculez la statistique de test $t_0$.
\begin{solution}
\[
t_0=\frac{\bar{x}-\mu_0}{s/\sqrt{n}}=\frac{104-100}{10/5}=\frac{4}{2}=2.
\]
\end{solution}

\part[3] Avec $t_{0{,}95;24}=1{,}711$, donnez la décision.
\begin{solution}
Région critique: rejeter $H_0$ si $t_0>1{,}711$.
Comme $2>1{,}711$, on rejette $H_0$.
\end{solution}

\part[3] Formulez la conclusion en contexte.
\begin{solution}
Au seuil de $5\%$, les données supportent que la moyenne du biomarqueur est supérieure à 100.
\end{solution}
\end{parts}

\question[8] ANOVA à 1 facteur - objectif: conclure avec la statistique $F$.
Une équipe compare 4 formulations d'un gel cutané sur un score de cicatrisation.
Le test ANOVA donne la table partielle suivante:
\begin{verbatim}
Source        SS      ddl      MS      F0
Traitement   48.0      3      16.0     4.0
Erreur       80.0     20       4.0
\end{verbatim}
\begin{parts}
\part[2] Écrivez les hypothèses du test global.
\begin{solution}
$H_0: \mu_1=\mu_2=\mu_3=\mu_4$.
$H_1$: au moins une moyenne diffère.
\end{solution}

\part[3] Si $F_{0{,}05;3,20}=3{,}10$, donnez la décision.
\begin{solution}
Comme $F_0=4{,}0>3{,}10$, on rejette $H_0$.
\end{solution}

\part[3] Interprétez en contexte.
\begin{solution}
Au seuil de $5\%$, il existe une différence significative entre les moyennes des 4 formulations.
\end{solution}
\end{parts}

\question[8] Régression linéaire simple - objectif: tester la pente.
On modélise la pression artérielle $Y$ en fonction de l'âge $X$.
On obtient $\hat{\beta}_1=0{,}80$, $SE(\hat{\beta}_1)=0{,}25$, avec $n=18$ observations.
On veut tester au seuil $5\%$:
\[
H_0:\beta_1=0\qquad\text{contre}\qquad H_1:\beta_1\neq 0.
\]
\begin{parts}
\part[3] Calculez la statistique de test.
\begin{solution}
\[
t_0=\frac{\hat{\beta}_1-0}{SE(\hat{\beta}_1)}=\frac{0{,}80}{0{,}25}=3{,}2.
\]
\end{solution}

\part[2] Avec $t_{0{,}025;16}=2{,}12$, donnez la décision.
\begin{solution}
Comme $|3{,}2|>2{,}12$, on rejette $H_0$.
\end{solution}

\part[3] Conclusion en contexte.
\begin{solution}
La relation linéaire entre la pression artérielle et l'âge est significative au seuil de $5\%$.
\end{solution}
\end{parts}

\question[8] Khi-carré d'ajustement - objectif: réaliser un test complet à partir d'un tableau.
On veut vérifier si les issues A, B, C, D sont équiprobables.
On observe 120 essais.

\begin{center}
\begin{tabular}{c|cccc|c}
Issue & A & B & C & D & Total\\
\hline
Fréquences observées $O_i$ & 24 & 36 & 30 & 30 & 120\\
Fréquences attendues $E_i$ (sous $H_0$) & 30 & 30 & 30 & 30 & 120
\end{tabular}
\end{center}

\begin{parts}
\part[3] Calculez la statistique $D_0$.
\begin{solution}
\[
D_0=\sum\frac{(O_i-E_i)^2}{E_i}
=\frac{(24-30)^2}{30}+\frac{(36-30)^2}{30}+\frac{(30-30)^2}{30}+\frac{(30-30)^2}{30}
=1{,}2+1{,}2=2{,}4.
\]
\end{solution}

\part[2] Donnez les degrés de liberté.
\begin{solution}
$\nu=k-1=4-1=3$.
\end{solution}

\part[3] Si $\chi^2_{0{,}05;3}=7{,}815$, concluez au seuil 5\%.
\begin{solution}
Comme $2{,}4<7{,}815$, on ne rejette pas $H_0$.
Les données sont compatibles avec des issues équiprobables.
\end{solution}
\end{parts}

\question[8] Khi-carré d'indépendance - objectif: conclure sur l'association.
On observe le tableau suivant:
\begin{center}
\begin{tabular}{c|cc|c}
 & Réussite & Échec & Total\\
\hline
Programme A & 32 & 18 & 50\\
Programme B & 20 & 30 & 50\\
\hline
Total & 52 & 48 & 100
\end{tabular}
\end{center}

\begin{parts}
\part[3] Calculez les fréquences attendues sous $H_0$.
\begin{solution}
$E_{A,R}=50\times52/100=26$, $E_{A,E}=24$,
$E_{B,R}=26$, $E_{B,E}=24$.
\end{solution}

\part[3] Calculez $D_0$.
\begin{solution}
\[
D_0=\frac{(32-26)^2}{26}+\frac{(18-24)^2}{24}+\frac{(20-26)^2}{26}+\frac{(30-24)^2}{24}
=1{,}385+1{,}5+1{,}385+1{,}5=5{,}77.
\]
\end{solution}

\part[2] Si $\chi^2_{0{,}05;1}=3{,}84$, donnez la conclusion.
\begin{solution}
Comme $5{,}77>3{,}84$, on rejette l'indépendance.
Il existe une association entre le programme et l'issue.
\end{solution}
\end{parts}
